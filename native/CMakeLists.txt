cmake_minimum_required(VERSION 3.20)
project(manga_translator_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
foreach(required ORT_ROOT LITERT_SDK WEBP_SOURCE_DIR)
  if(NOT DEFINED ${required})
    message(FATAL_ERROR "Set ${required} to a verified dependency location")
  endif()
endforeach()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_ANIM_UTILS OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_CWEBP OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_DWEBP OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_GIF2WEBP OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_IMG2WEBP OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_VWEBP OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_WEBPINFO OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_WEBPMUX OFF CACHE BOOL "" FORCE)
set(WEBP_BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
add_subdirectory("${WEBP_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/webp" EXCLUDE_FROM_ALL)
add_executable(manga-translator-server server.cpp)
target_include_directories(manga-translator-server PRIVATE "${ORT_ROOT}/include" "${LITERT_SDK}")
if(WIN32)
  enable_language(RC)
  target_sources(manga-translator-server PRIVATE windows.rc)
  target_compile_definitions(manga-translator-server PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(manga-translator-server PRIVATE webpdecoder ws2_32 crypt32 bcrypt psapi)
  if(MINGW)
    target_link_options(manga-translator-server PRIVATE -static -municode)
  endif()
else()
  find_package(Threads REQUIRED)
  target_link_libraries(manga-translator-server PRIVATE "${ORT_ROOT}/lib/libonnxruntime.so" webpdecoder Threads::Threads ${CMAKE_DL_LIBS})
  # The controller supplies the selected runtime directories at launch. Never
  # embed a developer machine's SDK path or a particular release folder.
  set_target_properties(manga-translator-server PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "")
endif()

# Match the published image preprocessing across CPU architectures.
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(manga-translator-server PRIVATE -ffp-contract=off)
endif()
